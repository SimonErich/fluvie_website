Create from assets

Create a gentle 4-second video about fictional cat Milo playing with a red ball. Use assets/ball.png and assets/play.mp4 and assets/music.wav. The image and clip are synthetic red geometric fixtures, not photos of a cat. Use a 160 by 160 canvas at 30 fps, two readable scenes and a short title in 18px text. Use only these asset keys. Use a size object with width 160 and height 160. Keep the result four seconds long.

Recorded workflow: 80.1 seconds · encoded output verified · selected frames repeatable.

The generated Dart
import 'package:flutter/material.dart' hide Animation, Clip, Image, Tween;
import 'package:fluvie/fluvie.dart';

Video build() {
  return Video(
    size: VideoSize(160, 160),
    audio: [Audio.music('assets/music.wav', volume: 0.4, fadeIn: 500.ms, fadeOut: 500.ms)],
    scenes: [
      Scene(
        duration: 1.5.seconds,
        background: Background.gradient(
          [Color(0xFF1A1A1A), Color(0xFF000000)],
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
        ),
        children: [
          ElementId(
            id: 'title_text',
            child: Placed(
              placement: Placement(x: 0.5, y: 0.5),
              child: Text(
                'Milo and the Red Ball',
                style: TextStyle(
                  color: Color(0xFFFFFFFF),
                  fontSize: 18,
                  fontWeight: FontWeight.bold,
                ),
              ).animate([Animation.fadeIn(duration: 0.8.seconds, delay: 0.3.seconds)]),
            ),
          ),
        ],
      ),
      Scene(
        duration: 2.5.seconds,
        background: Background.color(Color(0xFF1A1A1A)),
        children: [
          ElementId(
            id: 'milo_play_clip',
            child: Placed(
              placement: Placement(x: 0.5, y: 0.5, width: 0.9, height: 0.9),
              child: Clip.asset(
                'assets/play.mp4',
                fit: BoxFit.contain,
                audio: ClipAudio.muted(),
              ).animate([Animation.fadeIn(duration: 0.5.seconds)]),
            ),
          ),
          ElementId(
            id: 'red_ball_image',
            child: Placed(
              placement: Placement(x: 0.6, y: 0.6, width: 0.2, height: 0.2),
              child: Image.asset('assets/ball.png').animate([
                Animation.pop(duration: 0.4.seconds, delay: 0.2.seconds),
                Animation.drift(delay: 0.6.seconds),
              ]),
            ),
          ),
        ],
      ),
    ],
  );
}
